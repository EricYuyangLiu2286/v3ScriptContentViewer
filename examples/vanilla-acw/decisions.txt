rule ID1 boundary @Ideologies type=ideology # ideologies only state an attitude towards the countdown

in event:acw_events.1 # flavour event fired by the countdown's monthly pulse

in event:acw_events.3 # flavour event, monthly pulse
in event:acw_events.5 # flavour event, yearly pulse
in event:acw_events.6 # flavour event, yearly pulse

in event:slavery_law_events.7 # the success outcome: reconciliation
in var:acw_progress_var # countdown progress, set at start
in var:csa_will_spawn_var # set when the war is about to start
in var:je_acw_countdown_complete # set by the success outcome
in modifier:missouri_compromise # applied by the success outcome
in movesupport:movement_support_je_acw_countdown # movement pop support while the countdown runs
in svalue:movement_radicalism_acw_countdown # movement radicalism during the countdown

boundary party:conservative_party @Parties # party conditions read the countdown

boundary pbname:golden_circle @Power_bloc_names # a power bloc name that depends on the countdown
boundary history:countries/usa_-_usa @Start_history # the USA start history; drawn, not expanded
soft no *|*|law:* # laws are shared by many systems; not a theme link
